cmake_minimum_required(VERSION 3.20)

project(NXPKPacker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(lz4 CONFIG REQUIRED)

add_executable(NXPKPacker
    NXPKPacker.cpp
)

target_include_directories(NXPKPacker PRIVATE
    include
    
)

target_link_libraries(NXPKPacker PRIVATE
    lz4
)
